Bootstrap distributions as brain-wide pixel mosaics

This notebook generates Figure 1 from Getting over ANOVA: Estimation graphics for multi-group comparisons (Lu, Anns, Mai, Xu et al., 2026).

Each cell in a whorlmap encodes a full bootstrap distribution in 7x7 pixels. Pixels are filled via an outside-in spiral; position encodes quantile rank and colour encodes effect size.

Runtime: approx 15 min total (numba JIT compilation ~1 min; 260 bootstrap comparisons ~10-15 min)

Run order: Runtime -> Run all, or execute cells top-to-bottom.

def _spiralize(fill, m, n):    i = 0; j = 0; k = 0; array = np.zeros((m, n))    while m > 0 and k < len(fill):        jj = j; ii = i        for j in range(j, n):            if k >= len(fill): break            array[i, j] = fill[k]; k += 1        for i in range(ii + 1, m):            if k >= len(fill): break            array[i, j] = fill[k]; k += 1        for j in range(n - 2, jj - 1, -1):            if k >= len(fill): break            array[i, j] = fill[k]; k += 1        for i in range(m - 2, ii, -1):            if k >= len(fill): break            array[i, j] = fill[k]; k += 1        m -= 1; n -= 1; j += 1    return arraydef _spiral_path(n):    path = []    top, bot, lft, rgt = 0, n - 1, 0, n - 1    while top <= bot and lft <= rgt:        for c in range(lft, rgt + 1): path.append((top, c))        top += 1        for r in range(top, bot + 1): path.append((r, rgt))        rgt -= 1        if top <= bot:            for c in range(rgt, lft - 1, -1): path.append((bot, c))            bot -= 1        if lft <= rgt:            for r in range(bot, top - 1, -1): path.append((r, lft))            lft += 1    return pathdef _ring_structure(n):    rings, pos, k = [], 0, 0    while True:        sz = n - 2 * k        if sz <= 0: break        nc = 1 if sz == 1 else 4 * (sz - 1)        rings.append({'k': k, 'start': pos, 'end': pos + nc, 'sz': sz})        pos += nc; k += 1        if sz == 1: break    return rings
